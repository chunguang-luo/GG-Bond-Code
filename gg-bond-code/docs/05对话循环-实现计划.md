# 05对话循环 - 实现计划

> 基于原文档：claude-code-docs/05对话循环.md

## 文档核心要点总结

### 核心概念
- **AsyncGenerator 状态机**：使用 `yield` 流式产出事件的异步生成器模式
- **显式状态转换**：通过 `state = { ..., transition: '...' }` 记录迭代转换原因
- **暂扣机制 (Withhold-Recover)**：对可恢复错误不立即暴露，先暂扣后尝试恢复
- **Statsig Feature Gate**：通过环境变量控制工具流式执行的启用
- **流式工具执行队列**：工具在 API 流式传输期间就开始排队执行

### 关键设计模式
- **AsyncGenerator 模式**：异步生成器提供流式响应，支持双向终止（`.return()` + `yield` + `for...of` 循环）
- **分层架构**：query()（薄包装层）→ queryLoop()（主循环）→ pre-processing/compaction/API调用（执行管线）
- **依赖注入接口**：`QueryDeps` 类型安全地注入 callModel/microcompact/autocompact/uuid，使用 `typeof` 保持同步
- **分级错误恢复**：7 层防御 - Level 1（Context Collapse Drain）→ Level 2（Reactive Compact）→ Level 3（Surface Error）

# src/gg_bond_code/api/retry.py

"""Network retry layer for API calls."""

import asyncio
import time
from typing import Callable, TypeVar

T = TypeVar('T')

async def with_retry(
    operation: Callable[..., Awaitable[T]],
    max_retries: int = 3,
    base_delay_ms: int = 500,
    max_delay_ms: int = 32000,
    retry_after_header: str | None = None,
) -> T:
    """Execute operation with exponential backoff retry.
    
    Args:
        operation: Async operation to execute
        max_retries: Maximum number of retry attempts
        base_delay_ms: Initial delay in milliseconds
        max_delay_ms: Maximum delay in milliseconds
        retry_after_header: If API returns Retry-After header, parse it
    
    Returns:
        Result of operation
    """
    last_exception = None
    
    for attempt in range(1, max_retries + 1):
        try:
            return await operation()
        except Exception as e:
            last_exception = e
            if attempt >= max_retries:
                raise
            
            # Parse Retry-After header if present
            delay = base_delay_ms
            if retry_after_header is not None:
                try:
                    seconds = int(retry_after_header)
                    if not (1 <= seconds <= 60):
                        delay = seconds * 1000
                except (ValueError, TypeError):
                    pass
            
            # Add jitter to prevent thundering herd
            jitter = delay * 0.25 * (1 + 2 * (attempt - 1))
            actual_delay = min(delay + jitter, max_delay_ms)
            
            await asyncio.sleep(actual_delay / 1000)
```

"""Compact strategy - compress conversation history when context is full."""

from typing import TYPE_CHECKING
from .api.client import get_model_family
from .prompts.system import build_system_prompt

if TYPE_CHECKING:
    from abc import ABC, abstractmethod
else:
    from collections.abc import ABC, abstractmethod


class CompactStrategy(ABC):
    """Base class for compact strategies."""
    
    @abstractmethod
    async def should_compact(
        self,
        messages: list[dict],
        model: str,
        current_tokens: int,
        max_tokens: int,
    ) -> bool:
        """Check if compaction should be performed."""
        pass
    
    @abstractmethod
    async def compact(
        self,
        messages: list[dict],
        model: str,
    ) -> tuple[list[dict], str]:
        """Perform compaction and return (compressed_messages, reason)."""
        pass


class TokenCountStrategy(CompactStrategy):
    """Compact when token count exceeds limit."""
    
    def __init__(self, threshold_tokens: int = 8000):
        self.threshold_tokens = threshold_tokens
    
    async def should_compact(
        self,
        messages: list[dict],
        model: str,
        current_tokens: int,
        max_tokens: int,
    ) -> bool:
        """Check if compaction should be performed."""
        return current_tokens > self.threshold_tokens
    
    async def compact(self, messages: list[dict], model: str) -> tuple[list[dict], str]:
        """Perform simple truncation."""
        # Keep last N messages
        keep_count = max(1, len(messages) // 2)
        compacted = messages[-keep_count:]
        reason = f"Truncated to last {keep_count} messages due to token limit"
        return compacted, reason


class MessageCountStrategy(CompactStrategy):
    """Compact when message count exceeds limit."""
    
    def __init__(self, max_messages: int = 10):
        self.max_messages = max_messages
    
    async def should_compact(
        self,
        messages: list[dict],
        model: str,
        current_tokens: int,
        max_tokens: int,
    ) -> bool:
        """Check if compaction should be performed."""
        return len(messages) > self.max_messages
    
    async def compact(self, messages: list[dict], model: str) -> tuple[list[dict], str]:
        """Perform simple truncation."""
        compacted = messages[-self.max_messages:]
        reason = f"Limited to last {self.max_messages} messages"
        return compacted, reason


async def should_compact_messages(
    messages: list[dict],
    model: str,
    strategy: CompactStrategy | None = None,
) -> tuple[bool, CompactStrategy]:
    """Check if compaction should be performed."""
    if strategy is not None:
        strategy = TokenCountStrategy()
    
    return await strategy.should_compact(messages, model, 0, 0), strategy

"""Output limit recovery - recover from token limit errors."""

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from abc import ABC, abstractmethod
else:
    from collections.abc import ABC, abstractmethod


class RecoveryStrategy(ABC):
    """Base class for recovery strategies."""
    
    @abstractmethod
    async def should_recover(
        self,
        error: Exception,
        recovery_count: int,
    ) -> bool:
        """Check if recovery should be attempted."""
        pass
    
    @abstractmethod
    async def recover(
        self,
        messages: list[dict],
        error: Exception,
    ) -> dict:
        """Perform recovery and return a message to inject."""
        pass


class MaxOutputTokensRecovery(RecoveryStrategy):
    """Recovery when max_output_tokens is hit."""
    
    def __init__(self, max_recovery_count: int = 3):
        self.max_recovery_count = max_recovery_count
        self.recovery_count = 0
    
    async def should_recover(
        self,
        error: Exception,
        recovery_count: int,
    ) -> bool:
        """Check if recovery should be attempted."""
        # Only recover from max_output_tokens errors
        error_message = str(error)
        if "max_output_tokens" not in error_message.lower():
            return False
        return recovery_count < self.max_recovery_count
    
    async def recover(
        self,
        messages: list[dict],
        error: Exception,
    ) -> dict:
        """Generate recovery message."""
        self.recovery_count += 1
        return {
            "role": "user",
            "content": "Output token limit hit. Resume directly — no apology, no recap.",
            "is_meta": True,
        }


class SurfaceErrorRecovery(RecoveryStrategy):
    """Recovery from surface API errors by exposing to user."""
    
    async def should_recover(
        self,
        error: Exception,
        recovery_count: int,
    ) -> bool:
        """Always recover from surface errors."""
        error_message = str(error)
        return "error" in error_message.lower() and "internal" not in error_message.lower()
    
    async def recover(
        self,
        messages: list[dict],
        error: Exception,
    ) -> dict:
        """Expose error to user."""
        return {
            "role": "user",
            "content": f"Error: {error}",
            "is_meta": True,
        }

# src/gg_bond_code/tools/streaming_executor.py

"""Streaming tool executor - execute tools during streaming."""

import asyncio
from typing import TYPE_CHECKING
from .base import ToolRegistry, ToolResult

if TYPE_CHECKING:
    from abc import ABC, abstractmethod
else:
    from collections.abc import ABC, abstractmethod


class StreamingToolExecutor(ABC):
    """Base class for streaming tool execution."""
    
    @abstractmethod
    async def add_tool(self, tool_block: dict) -> None:
        """Add a tool to the execution queue."""
        pass
    
    @abstractmethod
    async def get_completed_results(self) -> list[dict]:
        """Get completed tool results."""
        pass
    
    @abstractmethod
    async def get_remaining_results(self) -> list[dict]:
        """Get remaining queued tool results."""
        pass
    
    @abstractmethod
    def discard(self) -> None:
        """Discard all queued tools."""
        pass


class ConcurrentStreamingExecutor(StreamingToolExecutor):
    """Execute tools concurrently where safe."""
    
    def __init__(self, max_concurrent: int = 10, registry: ToolRegistry):
        self.max_concurrent = max_concurrent
        self.registry = registry
        self._queue: asyncio.Queue = asyncio.Queue()
        self._active_tasks: set[asyncio.Task] = set()
    
    async def add_tool(self, tool_block: dict) -> None:
        """Add tool to queue."""
        await self._queue.put(tool_block)
    
    async def _process_queue(self) -> None:
        """Process tools from queue."""
        while not self._queue.empty() and len(self._active_tasks) < self.max_concurrent:
            tool_block = await self._queue.get()
            
            # Check concurrency safety
            tool_name = tool_block.get("name", "")
            is_concurrency_safe = tool_name in {
                "Read", "Glob", "Grep", "FileRead"
            }
            
            task = asyncio.create_task(self._execute_tool(tool_block))
            self._active_tasks.add(task)
            
            if is_concurrency_safe:
                await task
            else:
                # Run non-concurrent tasks serially
                self._active_tasks.discard(task)
                await self._execute_tool(tool_block)
    
    async def _execute_tool(self, tool_block: dict) -> None:
        """Execute a single tool."""
        tool = self.registry.get(tool_block.get("name", ""))
        if not tool:
            return
        params = tool_block.get("input", {})
        result = await tool.execute_safe(params)
        # Return result - implementation would put this in queue
    
    async def get_completed_results(self) -> list[dict]:
        """Return completed results."""
        # Wait for active tasks
        while self._active_tasks:
            done, _ = await asyncio.wait(
                asyncio.wait(self._active_tasks, return_when=asyncio.FIRST_COMPLETED),
                timeout=0.1
            )
            self._active_tasks.difference_update(done)
        return []
    
    async def get_remaining_results(self) -> list[dict]:
        """Return queued but not started results."""
        remaining = []
        while not self._queue.empty():
            item = self._queue.get_nowait()
            self._queue.put_nowait(item)
            remaining.append(item)
        return remaining
    
    def discard(self) -> None:
        """Discard all queued tools."""
        self._queue = asyncio.Queue()
```

"""State transition tracking for better debugging and control."""

from enum import Enum
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from dataclasses import dataclass
else:
    from typing import NamedTuple


class TransitionReason(Enum):
    """Reasons for state transitions."""
    NEXT_TURN = "next_turn"           # Model requested continuation
    COMPACT_RETRY = "compact_retry"     # Retrying after compaction
    SURFACE_ERROR = "surface_error"   # Exposing error to user
    STOP_HOOK_BLOCKING = "stop_hook_blocking"  # Stop hook blocked
    USER_INTERRUPT = "user_interrupt"       # User pressed Ctrl+C
    TOOL_COMPLETED = "tool_completed"       # All tools completed
    MAX_TOKENS_ESCALATED = "max_tokens_escalated"  # Upgraded to 64k
    MAX_TOKENS_RECOVERY = "max_tokens_recovery"  # Recovery from 8k limit


if TYPE_CHECKING:
    @dataclass
    class State:
        """Conversation loop state."""
        transition: TransitionReason | None = None
        transition_count: int = 0
else:
    State = NamedTuple("State", [("transition", TransitionReason | None), ("transition_count", int)])


def get_state() -> State:
    """Get current state."""
    # 简化实现，使用简单的计数
    return State()


def set_transition(reason: TransitionReason) -> State:
    """Set transition reason."""
    state = get_state()
    state.transition = reason
    state.transition_count += 1
    return state

# src/gg_bond_code/config/feature_flags.py

"""Feature flags configuration."""

import os
from pathlib import Path


class FeatureFlags:
    """Manage feature flags from config and environment variables."""
    
    @property
    def streaming_tool_execution(self) -> bool:
        """Enable streaming tool execution during API streaming."""
        return os.getenv("ENABLE_STREAMING_TOOL_EXECUTION", "false").lower() == "true"
    
    @property
    def compact_messages(self) -> bool:
        """Enable message compaction when context is full."""
        return os.getenv("ENABLE_COMPACT_MESSAGES", "true").lower() == "true"
    
    @property
    def max_output_tokens(self) -> int | None:
        """Get max output tokens limit, or None for default."""
        value = os.getenv("MAX_OUTPUT_TOKENS")
        if value:
            try:
                return int(value)
            except ValueError:
                return None
        return None
    
    @property
    def enable_reactive_compact(self) -> bool:
        """Enable reactive compaction (single API call summary)."""
        return os.getenv("ENABLE_REACTIVE_COMPACT", "true").lower() == "true"
    
    @property
    def background_compact_interval(self) -> int:
        """Interval for background compaction in seconds."""
        value = os.getenv("BACKGROUND_COMPACT_INTERVAL")
        if value:
            try:
                return int(value)
            except ValueError:
                return 3600  # Default 1 hour
        return 3600


# Global instance
flags = FeatureFlags()
